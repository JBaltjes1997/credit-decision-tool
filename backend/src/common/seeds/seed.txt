create table Wallet(
	walletID varchar(255) not null unique,
	account varchar(255),
	username varchar(255),
	primary key (walletId)
);

create table Limits(
	currentLimit bigint,
	proposedLimit bigint, 
	predictedLimit bigint,
	expectedLimit bigint,
	foreign key (account) references Wallet(account)
);

create table Details(
	walletAccount varchar(255) not null,
  	CRBScore varchar(255),
	estimatedIncome bigint,
  	salaryIn bigint,
  	totalVolumeIn bigint,
  	totalVolumeOut bigint,
  	wholesaleAvgOut bigint,
  	wholesaleMaxOut bigint,
  	creditIn bigint,
  	creditOut bigint,
  	fuliza bigint,
  	overdraft bigint,
  	liquorOut bigint,
  	gamblingOut bigint,
	primary key (walletAccount)
);

create table User(
	id varchar(255) not null unique,
	username varchar(255),
	phonenumber varchar(255),
	primary key (id),
	foreign key (account) references Wallet(account)
);

create table user to wallet(
	id varchar(255),
	name varchar(255),
	primary key (id)
);
